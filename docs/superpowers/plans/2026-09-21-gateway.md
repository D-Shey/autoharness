# Gateway-MCP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** autoharness registers itself in the agent once as an MCP server (`gateway serve`) and runs servers added with `add mcp` itself, so they are usable without restarting the agent.

**Architecture:** `mcp_client.py` talks to child MCP servers (stdio processes and Streamable HTTP). `gateway.py` is the MCP server the agent talks to: it watches `.autoharness/gateway.json`, starts/stops children, exposes two fixed meta-tools (`harness_tools`, `harness_call`) plus native `<server>__<tool>` tools, and sends `notifications/tools/list_changed`. `env.py` routes `add mcp` into `gateway.json` when the gateway is installed; `cli.py` adds `gateway install|uninstall|status|serve` and `add mcp --direct`.

**Tech Stack:** Python 3.10+, stdlib only (`subprocess`, `threading`, `urllib`, `http.server` in tests), pytest.

**Spec:** `docs/superpowers/specs/2026-09-21-gateway-design.md`

## Global Constraints

- No third-party runtime dependencies.
- Gateway stdout carries only JSON-RPC, one message per line, UTF-8, `\n` line endings. Logs go to stderr and `.autoharness/gateway.log`.
- `PROTOCOL_VERSIONS = ("2025-11-25", "2025-06-18", "2025-03-26", "2024-11-05")`; the first is ours.
- Gateway server name in the agent config: `autoharness`. Meta-tools: `harness_tools`, `harness_call`. Native tool name: `re.sub(r"[^a-zA-Z0-9_-]", "_", f"{server}__{tool}")[:64]`.
- Proxyable config: has `command`, or `type == "http"` with non-empty `headers`.
- Child request timeout: `AUTOHARNESS_GATEWAY_TIMEOUT` seconds, default 120, read at each request.
- Secrets stay `${VAR}` on disk; substituted from the gateway's environment at child start.
- Tests need no network and no Node. Run with `python -m pytest -q` from the repo root; all existing tests stay green.
- Import direction: `gateway` imports `env` and `mcp_client`; `env` never imports `gateway` or `mcp_client`.

```python
"""A tiny stdio MCP server for tests: echo, slow, add_tool (emits list_changed), crash, client_reply."""
import json
import sys
import time

TOOLS = {n: {"name": n, "description": d, "inputSchema": {"type": "object"}} for n, d in [
    ("echo", "Echo text back"), ("slow", "Sleep, then answer"), ("add_tool", "Add a tool and announce it"),
    ("crash", "Exit without answering"), ("client_reply", "What the client answered to our request")]}
state = {"reply": None}


def send(msg):
    sys.stdout.write(json.dumps(msg) + "\n")
    sys.stdout.flush()


def text(t):
    return {"content": [{"type": "text", "text": t}]}


while True:
    line = sys.stdin.readline()
    if not line:
        break
    msg = json.loads(line)
    if "method" not in msg:          # the client's answer to our roots/list request
        state["reply"] = msg
        continue
    method, mid = msg["method"], msg.get("id")
    if method == "initialize":
        send({"jsonrpc": "2.0", "id": mid, "result": {
            "protocolVersion": msg["params"]["protocolVersion"],
            "capabilities": {"tools": {"listChanged": True}}, "serverInfo": {"name": "fake", "version": "1"}}})
    elif method == "notifications/initialized":
        send({"jsonrpc": "2.0", "id": "srv-1", "method": "roots/list"})
    elif method == "tools/list":
        send({"jsonrpc": "2.0", "id": mid, "result": {"tools": list(TOOLS.values())}})
    elif method == "tools/call":
        name, args = msg["params"]["name"], msg["params"].get("arguments") or {}
        if name == "echo":
            result = text(args.get("text", ""))
        elif name == "slow":
            time.sleep(float(args.get("seconds", 5)))
            result = text("done")
        elif name == "add_tool":
            TOOLS["extra"] = {"name": "extra", "description": "Added later", "inputSchema": {"type": "object"}}
            send({"jsonrpc": "2.0", "method": "notifications/tools/list_changed"})
            result = text("added")
        elif name == "crash":
            sys.exit(1)
        elif name == "client_reply":
            result = text(json.dumps(state["reply"]))
        elif name == "extra":
            result = text("extra")
        else:
            send({"jsonrpc": "2.0", "id": mid, "error": {"code": -32602, "message": f"unknown tool {name}"}})
            continue
        send({"jsonrpc": "2.0", "id": mid, "result": result})
    elif mid is not None:
        send({"jsonrpc": "2.0", "id": mid, "error": {"code": -32601, "message": "not supported"}})
```

```python
"""Clients for the MCP servers the gateway runs on the agent's behalf: stdio processes and Streamable HTTP."""

from __future__ import annotations

import json
import os
import re
import shutil
import subprocess
import threading
from typing import Callable, Mapping

PROTOCOL_VERSIONS = ("2025-11-25", "2025-06-18", "2025-03-26", "2024-11-05")
VAR = re.compile(r"\$\{([A-Za-z_][A-Za-z0-9_]*)\}")


class McpError(Exception):
    pass


def timeout_s() -> float:
    return float(os.environ.get("AUTOHARNESS_GATEWAY_TIMEOUT", "120"))


def expand_secrets(value, env: Mapping[str, str] | None = None):
    """Replace ${VAR} with its value from `env` (default: os.environ), recursively."""
    env = os.environ if env is None else env

    def sub(m: re.Match) -> str:
        if m.group(1) not in env:
            raise McpError(f"set {m.group(1)} in the environment the agent is started from")
        return env[m.group(1)]

    if isinstance(value, str):
        return VAR.sub(sub, value)
    if isinstance(value, list):
        return [expand_secrets(v, env) for v in value]
    if isinstance(value, dict):
        return {k: expand_secrets(v, env) for k, v in value.items()}
    return value


def _nothing(*_):
    return None


class _Client:
    transport = ""

    def __init__(self, name: str, config: dict, log: Callable[[str], None] = _nothing,
                 on_tools_changed: Callable[[str], None] = _nothing, stderr=None):
        self.name, self.config, self.log, self.on_tools_changed = name, config, log, on_tools_changed
        self.stderr = stderr
        self.protocol_version = PROTOCOL_VERSIONS[0]
        self._id_lock = threading.Lock()
        self._next_id = 0

    def _new_id(self) -> int:
        with self._id_lock:
            self._next_id += 1
            return self._next_id

    def _handshake(self) -> None:
        res = self.request("initialize", {"protocolVersion": PROTOCOL_VERSIONS[0], "capabilities": {},
                                          "clientInfo": {"name": "autoharness-gateway", "version": "0.1"}})
        self.protocol_version = res.get("protocolVersion", self.protocol_version)
        self.notify("notifications/initialized")

    def list_tools(self) -> list[dict]:
        tools, cursor = [], None
        while True:
            res = self.request("tools/list", {"cursor": cursor} if cursor else {})
            tools += res.get("tools", [])
            cursor = res.get("nextCursor")
            if not cursor:
                return tools

    def call_tool(self, tool: str, arguments: dict | None) -> dict:
        return self.request("tools/call", {"name": tool, "arguments": arguments or {}})

    def _raise_for(self, method: str, msg: dict) -> dict:
        if "error" in msg:
            err = msg["error"]
            raise McpError(f"{self.name}: {err.get('message', err) if isinstance(err, dict) else err}")
        return msg.get("result", {})


class StdioClient(_Client):
    transport = "stdio"

    def start(self) -> None:
        cfg = expand_secrets(self.config)
        exe = shutil.which(cfg["command"])
        if not exe:
            raise McpError(f"command not found: {cfg['command']}")
        self._pending: dict[int, dict] = {}
        self._write_lock = threading.Lock()
        self.proc = subprocess.Popen([exe, *cfg.get("args", [])], stdin=subprocess.PIPE, stdout=subprocess.PIPE,
                                     stderr=self.stderr if self.stderr is not None else subprocess.DEVNULL,
                                     env={**os.environ, **cfg.get("env", {})})
        threading.Thread(target=self._read, daemon=True).start()
        self._handshake()

    def alive(self) -> bool:
        return getattr(self, "proc", None) is not None and self.proc.poll() is None

    def _send(self, msg: dict) -> None:
        data = (json.dumps(msg) + "\n").encode("utf-8")
        with self._write_lock:
            try:
                self.proc.stdin.write(data)
                self.proc.stdin.flush()
            except (BrokenPipeError, OSError, ValueError) as e:
                raise McpError(f"{self.name}: server is not running") from e

    def _read(self) -> None:
        for raw in self.proc.stdout:
            try:
                msg = json.loads(raw.decode("utf-8"))
            except ValueError:
                self.log(f"{self.name}: non-JSON output: {raw[:200]!r}")
                continue
            if "method" not in msg and msg.get("id") in self._pending:
                slot = self._pending.pop(msg["id"])
                slot["msg"] = msg
                slot["event"].set()
            elif "method" in msg and "id" in msg:
                try:
                    self._send({"jsonrpc": "2.0", "id": msg["id"],
                                "error": {"code": -32601, "message": f"autoharness gateway does not support {msg['method']}"}})
                except McpError:
                    pass
            elif msg.get("method") == "notifications/tools/list_changed":
                threading.Thread(target=self.on_tools_changed, args=(self.name,), daemon=True).start()
        for rid in list(self._pending):
            slot = self._pending.pop(rid, None)
            if slot:
                slot["msg"] = {"error": {"message": "server exited"}}
                slot["event"].set()

    def request(self, method: str, params: dict | None = None) -> dict:
        if not self.alive():
            raise McpError(f"{self.name}: server is not running")
        rid = self._new_id()
        slot = {"event": threading.Event(), "msg": None}
        self._pending[rid] = slot
        self._send({"jsonrpc": "2.0", "id": rid, "method": method, "params": params or {}})
        if not slot["event"].wait(timeout_s()):
            self._pending.pop(rid, None)
            raise McpError(f"{self.name}: {method} timed out after {timeout_s():g} s")
        return self._raise_for(method, slot["msg"])

    def notify(self, method: str, params: dict | None = None) -> None:
        self._send({"jsonrpc": "2.0", "method": method, **({"params": params} if params else {})})

    def close(self) -> None:
        proc = getattr(self, "proc", None)
        if proc is None or proc.poll() is not None:
            return
        try:
            proc.stdin.close()
        except OSError:
            pass
        try:
            proc.wait(timeout=3)
        except subprocess.TimeoutExpired:
            proc.kill()
            proc.wait(timeout=3)


def make_client(name: str, config: dict, log: Callable[[str], None] = _nothing,
                on_tools_changed: Callable[[str], None] = _nothing, stderr=None) -> _Client:
    if "command" in config:
        return StdioClient(name, config, log, on_tools_changed, stderr)
    raise McpError(f"{name}: unsupported transport; only stdio and http servers go through the gateway")
```

```python
class HttpClient(_Client):
    transport = "http"

    def start(self) -> None:
        self._cfg = expand_secrets(self.config)
        self.session: str | None = None
        self._handshake()

    def alive(self) -> bool:
        return True

    def _post(self, msg: dict) -> dict | None:
        headers = {"Content-Type": "application/json", "Accept": "application/json, text/event-stream",
                   **self._cfg.get("headers", {})}
        if self.session:
            headers["Mcp-Session-Id"] = self.session
        if msg.get("method") != "initialize":
            headers["MCP-Protocol-Version"] = self.protocol_version
        req = urllib.request.Request(self._cfg["url"], data=json.dumps(msg).encode("utf-8"),
                                     headers=headers, method="POST")
        try:
            resp = urllib.request.urlopen(req, timeout=timeout_s())
        except urllib.error.HTTPError as e:
            raise McpError(f"{self.name}: HTTP {e.code} {e.reason}") from e
        except OSError as e:
            raise McpError(f"{self.name}: {e}") from e
        with resp:
            self.session = resp.headers.get("Mcp-Session-Id") or self.session
            if "id" not in msg or resp.status == 202:
                return None
            if resp.headers.get("Content-Type", "").startswith("text/event-stream"):
                data: list[str] = []
                for raw in resp:
                    line = raw.decode("utf-8").rstrip("\r\n")
                    if line.startswith("data:"):
                        data.append(line[5:].lstrip())
                    elif not line and data:
                        event = json.loads("\n".join(data))
                        data = []
                        if event.get("id") == msg["id"] and ("result" in event or "error" in event):
                            return event
                raise McpError(f"{self.name}: event stream ended without a response")
            return json.loads(resp.read().decode("utf-8"))

    def request(self, method: str, params: dict | None = None) -> dict:
        msg = self._post({"jsonrpc": "2.0", "id": self._new_id(), "method": method, "params": params or {}})
        return self._raise_for(method, msg or {})

    def notify(self, method: str, params: dict | None = None) -> None:
        self._post({"jsonrpc": "2.0", "method": method, **({"params": params} if params else {})})

    def close(self) -> None:
        if getattr(self, "session", None):
            req = urllib.request.Request(self._cfg["url"], method="DELETE",
                                         headers={"Mcp-Session-Id": self.session, **self._cfg.get("headers", {})})
            try:
                urllib.request.urlopen(req, timeout=5).close()
            except OSError:
                pass
```

```python
"""The autoharness gateway: one MCP server the agent registers once, which runs the servers
`autoharness add mcp` puts in .autoharness/gateway.json, so they work without restarting the agent.

Two fixed meta-tools (harness_tools, harness_call) work in every agent. Native `<server>__<tool>`
tools and notifications/tools/list_changed are offered too, for agents that handle them.
"""

from __future__ import annotations

import json
import os
import re
import sys
import threading
import time
from pathlib import Path
from typing import Callable

from autoharness import __version__
from autoharness.mcp_client import PROTOCOL_VERSIONS, McpError, make_client

INSTRUCTIONS = ("Servers added with `autoharness add mcp` appear here without a restart. Call harness_tools to "
                "see them and their tools, then harness_call(server, tool, arguments). If your client shows "
                "tools named <server>__<tool>, you can call those directly.")
META_TOOLS = [
    {"name": "harness_tools",
     "description": "List MCP servers autoharness added for this task and their tools (with input schemas). "
                    "Call this after `autoharness add mcp`.",
     "inputSchema": {"type": "object", "properties": {"server": {"type": "string"}}}},
    {"name": "harness_call",
     "description": "Call a tool of an MCP server autoharness added: harness_call(server, tool, arguments).",
     "inputSchema": {"type": "object", "properties": {"server": {"type": "string"}, "tool": {"type": "string"},
                                                      "arguments": {"type": "object"}},
                     "required": ["server", "tool"]}},
]
LIST_CHANGED = {"jsonrpc": "2.0", "method": "notifications/tools/list_changed"}


def find_project(start: Path) -> Path | None:
    for d in [start, *start.parents]:
        if (d / ".autoharness" / "gateway.json").exists():
            return d
    return None


def _text(t: str, error: bool = False) -> dict:
    return {"content": [{"type": "text", "text": t}], **({"isError": True} if error else {})}


def native_name(server: str, tool: str) -> str:
    return re.sub(r"[^a-zA-Z0-9_-]", "_", f"{server}__{tool}")[:64]


class Gateway:
    def __init__(self, project: Path | None, write: Callable[[dict], None], log: Callable[[str], None],
                 stderr=None, client_factory=make_client):
        self.project, self.write, self.log, self.stderr = project, write, log, stderr
        self.client_factory = client_factory
        self.configs: dict[str, dict] = {}
        self.clients: dict[str, object] = {}
        self.status: dict[str, dict] = {}
        self.tools: dict[str, list[dict]] = {}
        self.registry_error: str | None = None
        self.initialized = False
        self._mtime: int | None = None
        self._lock = threading.RLock()

    # --- registry and children -------------------------------------------------

    def _registry_file(self) -> Path | None:
        return self.project / ".autoharness" / "gateway.json" if self.project else None

    def sync(self) -> bool:
        """Bring running servers in line with gateway.json. Returns whether anything changed."""
        path = self._registry_file()
        if path is None or not path.exists():
            if not self.configs:
                return False
            self._mtime, wanted = None, {}
        else:
            mtime = path.stat().st_mtime_ns
            if mtime == self._mtime:
                return False
            self._mtime = mtime
            try:
                wanted = json.loads(path.read_text("utf-8")).get("servers", {})
                self.registry_error = None
            except ValueError as e:
                self.registry_error = f"gateway.json is not valid JSON ({e}); keeping the previous servers"
                self.log(self.registry_error)
                return False
        changed = False
        with self._lock:
            for name in list(self.configs):
                if wanted.get(name) != self.configs[name]:
                    self._stop(name)
                    changed = True
            for name, cfg in wanted.items():
                if name not in self.configs:
                    self.configs[name] = cfg
                    self.status[name] = {"status": "starting"}
                    self.tools[name] = []
                    threading.Thread(target=self._boot, args=(name, cfg), daemon=True).start()
                    changed = True
        if changed:
            self._changed()
        return changed

    def _boot(self, name: str, cfg: dict) -> None:
        client = None
        try:
            client = self.client_factory(name, cfg, log=self.log, on_tools_changed=self._refresh, stderr=self.stderr)
            client.start()
            tools = client.list_tools()
        except (McpError, OSError) as e:
            if client is not None:
                client.close()
            with self._lock:
                if self.configs.get(name) is cfg:
                    self.status[name] = {"status": "error", "error": str(e)}
            self.log(f"{name}: {e}")
            self._changed()
            return
        with self._lock:
            if self.configs.get(name) is not cfg:
                client.close()
                return
            self.clients[name], self.tools[name], self.status[name] = client, tools, {"status": "running"}
        self._changed()

    def _stop(self, name: str) -> None:
        client = self.clients.pop(name, None)
        for d in (self.configs, self.status, self.tools):
            d.pop(name, None)
        if client is not None:
            threading.Thread(target=client.close, daemon=True).start()

    def _restart(self, name: str):
        with self._lock:
            cfg = self.configs.get(name)
            old = self.clients.pop(name, None)
            if cfg is None:
                return None
            self.status[name] = {"status": "starting"}
        if old is not None:
            old.close()
        self._boot(name, cfg)
        return self.clients.get(name)

    def _refresh(self, name: str) -> None:
        client = self.clients.get(name)
        if client is None:
            return
        try:
            tools = client.list_tools()
        except McpError as e:
            self.log(f"{name}: refreshing tools failed: {e}")
            return
        with self._lock:
            self.tools[name] = tools
        self._changed()

    def _changed(self) -> None:
        self._write_state()
        if self.initialized:
            self.write(LIST_CHANGED)

    def _write_state(self) -> None:
        if not self.project:
            return
        with self._lock:
            servers = {n: {**s, "tools": len(self.tools.get(n, []))} for n, s in self.status.items()}
        path = self.project / ".autoharness" / "gateway.state.json"
        try:
            path.write_text(json.dumps({"pid": os.getpid(), "updated": time.strftime("%Y-%m-%dT%H:%M:%S"),
                                        "servers": servers}, indent=2, ensure_ascii=False) + "\n", "utf-8")
        except OSError as e:
            self.log(f"could not write {path}: {e}")

    def close_all(self) -> None:
        with self._lock:
            clients = list(self.clients.values())
            self.clients.clear()
        for c in clients:
            c.close()

    # --- tools -----------------------------------------------------------------

    def native_map(self) -> dict[str, tuple[str, str]]:
        out: dict[str, tuple[str, str]] = {}
        with self._lock:
            for server in sorted(self.tools):
                for t in self.tools[server]:
                    out.setdefault(native_name(server, t["name"]), (server, t["name"]))
        return out

    def _tool_list(self) -> list[dict]:
        listed = list(META_TOOLS)
        with self._lock:
            by_key = {(s, t["name"]): t for s, ts in self.tools.items() for t in ts}
        for native, key in self.native_map().items():
            t = by_key[key]
            listed.append({"name": native, "description": f"[{key[0]}] {t.get('description', '')}".strip(),
                           "inputSchema": t.get("inputSchema") or {"type": "object"}})
        return listed

    def _describe(self, only: str | None) -> dict:
        reverse = {v: k for k, v in self.native_map().items()}
        servers = {}
        with self._lock:
            for name in sorted(self.status):
                if only and name != only:
                    continue
                st = dict(self.status[name])
                client = self.clients.get(name)
                if st["status"] == "running" and client is not None and not client.alive():
                    st = {"status": "error", "error": "the server process exited; the next call restarts it once"}
                servers[name] = {**st, "transport": "http" if self.configs.get(name, {}).get("type") == "http" else "stdio",
                                 "tools": [{"name": t["name"], "native_name": reverse.get((name, t["name"])),
                                            "description": t.get("description", ""),
                                            "inputSchema": t.get("inputSchema") or {"type": "object"}}
                                           for t in self.tools.get(name, [])]}
        out = {"project": str(self.project) if self.project else None, "servers": servers}
        if self.registry_error:
            out["registry_error"] = self.registry_error
        if not self.project:
            out["note"] = "no .autoharness/gateway.json found from the gateway's working directory upwards"
        return out

    def _forward(self, server: str | None, tool: str | None, arguments: dict) -> dict:
        with self._lock:
            client, st = self.clients.get(server), self.status.get(server)
        if client is None:
            if st:
                return _text(f"server {server!r} is {st['status']}" + (f": {st['error']}" if st.get("error") else ""), True)
            return _text(f"unknown server {server!r}; call harness_tools to see what is available", True)
        if not client.alive():
            client = self._restart(server)
            if client is None:
                return _text(f"server {server!r} stopped and did not restart: {self.status.get(server, {}).get('error')}", True)
        try:
            return client.call_tool(tool, arguments)
        except McpError as e:
            return _text(str(e), True)

    def _call(self, params: dict) -> dict:
        name, args = params.get("name"), params.get("arguments") or {}
        if name == "harness_tools":
            return _text(json.dumps(self._describe(args.get("server")), ensure_ascii=False, indent=2))
        if name == "harness_call":
            return self._forward(args.get("server"), args.get("tool"), args.get("arguments") or {})
        target = self.native_map().get(name)
        if target is None:
            return _text(f"unknown tool {name!r}; call harness_tools to see what is available", True)
        return self._forward(target[0], target[1], args)

    # --- JSON-RPC --------------------------------------------------------------

    def handle(self, msg: dict) -> dict | None:
        method, rid = msg.get("method"), msg.get("id")
        if method is None:
            return None
        if rid is None:
            if method == "notifications/initialized":
                self.initialized = True
            return None
        try:
            if method == "initialize":
                asked = msg.get("params", {}).get("protocolVersion")
                result = {"protocolVersion": asked if asked in PROTOCOL_VERSIONS else PROTOCOL_VERSIONS[0],
                          "capabilities": {"tools": {"listChanged": True}},
                          "serverInfo": {"name": "autoharness", "version": __version__},
                          "instructions": INSTRUCTIONS}
            elif method == "ping":
                result = {}
            elif method == "tools/list":
                result = {"tools": self._tool_list()}
            elif method == "tools/call":
                result = self._call(msg.get("params", {}))
            else:
                return {"jsonrpc": "2.0", "id": rid, "error": {"code": -32601, "message": f"method not found: {method}"}}
        except Exception as e:  # never let one request kill the gateway
            self.log(f"{method} failed: {e!r}")
            return {"jsonrpc": "2.0", "id": rid, "error": {"code": -32603, "message": str(e)}}
        return {"jsonrpc": "2.0", "id": rid, "result": result}
```

```python
def serve(start: Path | None = None) -> int:
    """Run the gateway on stdin/stdout until the agent closes stdin."""
    for stream in (sys.stdin, sys.stdout):
        if hasattr(stream, "reconfigure"):
            stream.reconfigure(encoding="utf-8", newline="\n")
    project = find_project((start or Path.cwd()).resolve())
    log_file = open(project / ".autoharness" / "gateway.log", "a", encoding="utf-8") if project else None
    out_lock = threading.Lock()

    def write(msg: dict) -> None:
        with out_lock:
            sys.stdout.write(json.dumps(msg, ensure_ascii=False) + "\n")
            sys.stdout.flush()

    def log(line: str) -> None:
        stamped = f"{time.strftime('%Y-%m-%d %H:%M:%S')} {line}"
        print(stamped, file=sys.stderr, flush=True)
        if log_file:
            log_file.write(stamped + "\n")
            log_file.flush()

    gw = Gateway(project, write, log, stderr=log_file)
    gw.sync()
    stop = threading.Event()

    def watch() -> None:
        while not stop.wait(1.0):
            try:
                gw.sync()
            except Exception as e:
                log(f"sync failed: {e!r}")

    threading.Thread(target=watch, daemon=True).start()
    try:
        for line in sys.stdin:
            line = line.strip()
            if not line:
                continue
            try:
                msg = json.loads(line)
            except ValueError:
                write({"jsonrpc": "2.0", "id": None, "error": {"code": -32700, "message": "parse error"}})
                continue
            if msg.get("method") == "tools/call":      # calls can be slow: do not block pings and lists
                threading.Thread(target=lambda m=msg: write(gw.handle(m)), daemon=True).start()
                continue
            resp = gw.handle(msg)
            if resp is not None:
                write(resp)
    finally:
        stop.set()
        gw.close_all()
        if log_file:
            log_file.close()
    return 0
```
